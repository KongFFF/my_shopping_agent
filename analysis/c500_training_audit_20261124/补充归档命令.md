# C500补充归档（无卡执行，不重训、不重跑CPU测试）

本次返回未包含run.json所指原生CPU收据，也未包含首次磁盘预检失败原现场。以下只读取既有文件，通过独立Git分支补回；缺失项照实记录，不把缺失当零更新证明。

```bash
cd /root/autodl-tmp/work/my_shopping_agent
bash <<'BASH'
set -euo pipefail
ROOT=/root/autodl-tmp/work/my_shopping_agent
PY="$ROOT/upstream/shopping-grpo-longhorizon/.venv/bin/python"
IFS= read -r RUN < "$ROOT/outputs/c500_active_run.txt"
DEST=$(mktemp -d "$ROOT/outputs/c500-supplement-XXXXXX")
git -c http.version=HTTP/1.1 clone --depth 1 --single-branch --filter=blob:none --sparse --branch codex/c500-gamma4 "$(git remote get-url origin)" "$DEST"
git -C "$DEST" sparse-checkout set analysis/c500_training_supplement
STAMP=$(date -u +%Y%m%dT%H%M%SZ)
BRANCH="codex/c500-supplement-$STAMP-$RANDOM"
git -C "$DEST" switch -c "$BRANCH"
REL="analysis/c500_training_supplement/$STAMP"
"$PY" - "$RUN" "$DEST/$REL" <<'PY'
from pathlib import Path
import json,sys,shutil,hashlib
run=Path(sys.argv[1]);out=Path(sys.argv[2]);out.mkdir(parents=True)
spec=json.loads((run/'run.json').read_text());assert spec['experiment']=='C500' and spec['gamma']==4
missing=[];files={}
def copy(src,name):
    if not src.is_file():missing.append(str(src));return
    dst=out/name;dst.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(src,dst)
    files[name]=hashlib.sha256(dst.read_bytes()).hexdigest()
copy(run/'run.json','run.json')
copy(Path(spec['cpu_receipt']),'native_cpu_acceptance.json')
copy(run/'zero_update_recovery.json','zero_update_recovery.json')
rec=json.loads((run/'zero_update_recovery.json').read_text())
old=Path(rec['previous_run']);assert old.parent==run.parent and old!=run
for name in ['run.json','stopped.json','supervisor.log','generation_budget.json','training_diagnostics.jsonl','weights.jsonl']:
    copy(old/name,'prior/'+name)
prior={'checkpoint_names':[p.name for p in (old/'train').glob('global_step_*')]}
for name in ['training_diagnostics.jsonl','weights.jsonl']:
    p=old/name
    if p.is_file():
        rows=[json.loads(x)for x in p.read_text().splitlines()if x.strip()]
        prior[name]={'rows':len(rows),'optimizer_events':sum(x.get('event')=='optimizer_step'for x in rows),'actor_update_events':sum(x.get('event','').startswith('actor_update_')for x in rows)}
(out/'manifest.json').write_text(json.dumps({'run':str(run),'files_sha256':files,'missing':missing,'prior_inventory':prior,'GPU_used':False,'training_started':False},indent=2)+'\n')
print('MISSING=',missing)
PY
git -C "$DEST" config user.name 'C500 Evidence Export'
git -C "$DEST" config user.email 'c500-evidence@localhost'
git -C "$DEST" add -- "$REL"
git -C "$DEST" commit -m 'Return existing C500 CPU receipt and preflight failure evidence'
COMMIT=$(git -C "$DEST" rev-parse HEAD)
git -C "$DEST" -c http.version=HTTP/1.1 push -u origin "$BRANCH"
REMOTE=$(git -C "$DEST" ls-remote origin "refs/heads/$BRANCH")
[[ "${REMOTE%%[[:space:]]*}" == "$COMMIT" ]]
printf 'RETURN_BRANCH=%s\nRETURN_COMMIT=%s\n' "$BRANCH" "$COMMIT"
BASH
```
